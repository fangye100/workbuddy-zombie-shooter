04 · 描边规则

  • 方法：inverted hull（背面外扩），不用后处理边缘检测——后者在顶视角密集尸潮时会糊成一团。
  • 线宽存储：顶点色 R 通道，0–1 映射 0–8 mm 世界空间外扩量，逐资产可微调。
  • 外扩量必须做摄像机距离补偿，保证屏幕空间线宽恒定（近处细远处粗会瞬间出戏）。
  • cullMode: 'front'，depth bias 微推，避免与本体 z-fighting。
  • 描边不接光照、不接 fog，纯 ink 色。禁止用阴影代替描边。
  • 顶视角特例：角色头顶与地面的夹角处容易出现断线，建模时头顶法线需手动外扩修形。

05 · 半调网点

  • 屏幕空间后处理，只叠加在暗部（luminance < 0.45），亮部保持干净——全屏铺网点会让画面变脏、移动端还会在细密纹理上产生摩尔纹。
  • 1080p 下网点尺寸 4–6 px，旋转 45° 打散网格感；移动端按 DPR 缩放，不要固定像素。
  • 强度控制在 0.15–0.25，超过 0.3 会和描边抢视觉。

06 · 投影与控件

  • 投影一律硬边偏移 6 px（shadow-hard），零模糊。模糊阴影会破坏漫画感。
  • UI 圆角：卡片 18 px / 面板 24 px；主按钮描边 4 px，强调按钮 6 px。
  • 按钮按下 = 位移 6 px + 投影归零，这是全游戏唯一的"物理反馈"，别加缩放动画。
  • 稀有度用卡片顶部 6 px 色条表达，不用整卡染色——同屏 12 张卡时整卡染色会糊成一片。

08 · 引擎落地（WGSL / Aether）

对应管线文档中的 Pass 拓扑：描边跟在 [8 Opaque] 之后、[14 Transparent] 之前；半调 + LUT 插在 [15 Post] 的 Tonemap 之后、Grading 之前。

描边 Pass · inverted hull

// 线宽来自顶点色 R 通道；外扩量随距离线性放大，保证屏幕空间线宽恒定
@vertex
fn vs_outline(
  @location(0) position: vec3f,
  @location(1) normal:   vec3f,
  @location(2) color:    vec4f,   // R = 线宽 0..1 → 0..8mm
) -> @builtin(position) vec4f {
  let worldPos = (model * vec4f(position, 1.0)).xyz;
  let worldNrm = normalize((model * vec4f(normal, 0.0)).xyz);
  let dist     = length(camera.position - worldPos);

  let width = color.r * 8.0 * 0.001 * dist * camera.pxToWorldScale;
  return camera.viewProj * vec4f(worldPos + worldNrm * width, 1.0);
}
// pipeline: cullMode = 'front'，depthBias 微推，避免与本体 z-fighting

@fragment
fn fs_outline() -> @location(0) vec4f {
  return vec4f(INK, 1.0);   // 纯墨线，不接光照、不接雾
}

2 阶 Toon 分阶

fn boostSat(c: vec3f, k: f32) -> vec3f {
  let l = dot(c, vec3f(0.2126, 0.7152, 0.0722));
  return clamp(l + (c - vec3f(l)) * k, vec3f(0.0), vec3f(1.0));
}

fn toonShade(albedo: vec3f, ndotl: f32, isWet: f32) -> vec3f {
  // 暗部：先压到 0.55，再把色相往 night-deep 拉，而不是压成灰
  let shadowCol = mix(boostSat(albedo * 0.55, 1.05), NIGHT_DEEP, 0.30);
  let litCol    = boostSat(albedo, 1.12);

  var c = select(shadowCol, litCol, ndotl > 0.30);
  // 高光层只给金属 / 黏液 / 湿润表面
  c = mix(c, mix(litCol * 1.06, BONE, 0.35), isWet * step(0.88, ndotl));
  return c;
}

let rot = mat2x2f(0.7071, -0.7071, 0.7071, 0.7071);  // 45° 打散网格感
let p   = rot * (frag.xy / dotSize);              // 1080p: dotSize = 4..6

let lum = dot(color, vec3f(0.2126, 0.7152, 0.0722));
let dot_ = 1.0 - smoothstep(-0.06, 0.06, d);
